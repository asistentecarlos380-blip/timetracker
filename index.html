async function cargarPanelTrabajador() { if (!currentUser) return; try { const res = await fetch('/api/mis-horas', { method: 'POST', headers: {'Content-Type': 'application/json'}, body: JSON.stringify({ usuario: currentUser.usuario, pin: currentUser.pin }) }); const data = await res.json(); registrosGlobalesTrabajador = data.registros || []; const tbody = document.getElementById('tabla-mis-horas'); if (!tbody) return; tbody.innerHTML = ''; if (registrosGlobalesTrabajador.length > 0) { registrosGlobalesTrabajador.forEach(r => { const tr = document.createElement('tr'); tr.className = 'border-b border-slate-700/40 hover:bg-slate-800/40'; const fechaStr = r.entrada ? r.entrada.split(' ')[0] : ''; const salidaTxt = r.salida ? r.salida : 'En Curso'; tr.innerHTML = ` ${fechaStr} ${r.entrada || ''} ${salidaTxt} - - `; tbody.appendChild(tr); }); } else { tbody.innerHTML = 'No hay registros de horas aún.'; } } catch (e) { console.error("Error al cargar horas del trabajador", e); } }
