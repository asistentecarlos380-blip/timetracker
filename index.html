function calcularResumenFechas() { if(!globalAdminData) return; const fDesde = document.getElementById('filtro-fecha-desde').value; const fHasta = document.getElementById('filtro-fecha-hasta').value; if(!fDesde || !fHasta) { alert("Por favor selecciona ambas fechas (Desde y Hasta)."); return; } let dDesde = new Date(fDesde + "T00:00:00"); let dHasta = new Date(fHasta + "T23:59:59"); let resumen = {}; for(let u in globalAdminData.usuarios) { resumen[u] = { tipo_pago: globalAdminData.usuarios[u].tipo_pago || 'Horas', monto_fijo: globalAdminData.usuarios[u].monto_fijo || 0, tarifa_hora: globalAdminData.usuarios[u].tarifa_hora !== undefined ? globalAdminData.usuarios[u].tarifa_hora : 1.5, rol_sistema: globalAdminData.usuarios[u].rol_sistema || 'trabajador', horas: 0 }; } globalAdminData.asistencias.forEach(r => { let dEntrada = parseFechaHora(r.entrada); let dSalida = parseFechaHora(r.salida); if(dEntrada && dEntrada >= dDesde && dEntrada <= dHasta) { let hrs = 0; if(dSalida) { hrs = Math.max(0, (dSalida - dEntrada) / (1000 * 60 * 60)); } if(!resumen[r.usuario]) { resumen[r.usuario] = { tipo_pago: 'Horas', monto_fijo: 0, tarifa_hora: 1.5, rol_sistema: 'trabajador', horas: 0 }; } resumen[r.usuario].horas += hrs; } }); const tR = document.getElementById('tabla-resumen-fechas'); tR.innerHTML = ''; for(let [u, info] of Object.entries(resumen)) { let totalPagar = 0; let tipoLower = info.tipo_pago.toLowerCase(); let valorTarifaTexto = ""; if(info.rol_sistema === 'admin') { totalPagar = 0; valorTarifaTexto = "Exento (Admin)"; } else if(tipoLower === 'gasolina') { totalPagar = info.horas * 1.4 * 0.5; valorTarifaTexto = "1.4L/h × $0.5"; } else if(tipoLower === 'fijo') { totalPagar = info.monto_fijo; valorTarifaTexto = `$${info.monto_fijo}`; } else { totalPagar = info.horas * info.tarifa_hora; valorTarifaTexto = `$${info.tarifa_hora}`; } tR.innerHTML += ` ${u} ${info.tipo_pago} ${info.horas.toFixed(2)} hrs ${valorTarifaTexto} $${totalPagar.toFixed(2)} `; } }
